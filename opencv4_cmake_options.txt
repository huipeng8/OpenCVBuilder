# =============================
# OpenCV 4.x CMake Options (Base)
# Shared by both /MD and /MT builds
# =============================

# World module (single library)
-DBUILD_opencv_world=ON

# Disable Python
-DBUILD_opencv_python2=OFF
-DBUILD_opencv_python3=OFF

# Extra modules (will be enabled/disabled per CRT mode in PS script)
-DOPENCV_EXTRA_MODULES_PATH=../opencv_contrib/modules

# Image codecs (essential)
-DBUILD_ZLIB=ON
-DBUILD_JPEG=ON -DWITH_JPEG=ON
-DBUILD_PNG=ON -DWITH_PNG=ON
-DBUILD_TIFF=ON -DWITH_TIFF=ON
-DBUILD_WEBP=ON -DWITH_WEBP=ON
-DBUILD_OPENEXR=ON -DWITH_OPENEXR=ON
-DBUILD_JASPER=ON -DWITH_JASPER=ON
-DWITH_IMGCODEC_GIF=ON
-DWITH_IMGCODEC_HDR=ON
-DWITH_IMGCODEC_PXM=ON
-DWITH_IMGCODEC_SUNRASTER=ON

# Core modules (required for world)
-DBUILD_opencv_core=ON
-DBUILD_opencv_imgproc=ON
-DBUILD_opencv_imgcodecs=ON
-DBUILD_opencv_features2d=ON
-DBUILD_opencv_calib3d=ON
-DBUILD_opencv_flann=ON

# Build control
-DBUILD_TESTS=OFF
-DBUILD_EXAMPLES=OFF
-DBUILD_DOCS=OFF
-DBUILD_PERF_TESTS=OFF
-DBUILD_FAT_JAVA_LIB=OFF
